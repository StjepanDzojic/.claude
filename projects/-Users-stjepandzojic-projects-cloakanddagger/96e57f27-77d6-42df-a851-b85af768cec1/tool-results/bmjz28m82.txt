+  if (isIpad()) {
+    const route = NavigationContainerRef.getCurrentRoute();
+    if (route?.name !== "home") NavigationContainerRef.navigate("home");
+    getDefaultStore().set(RoomAtoms.selectedRoomId, roomId);
+    return;
+  }
+
+  NavigationContainerRef.navigate("room", { roomId });
+};
+
+/** Reset the navigation stack so the user lands on the room. On iPad the stack
+ * is reset to home and the room is shown in the right pane via the selected
+ * room atom. */
+export const resetToRoom = (roomId: string) => {
+  if (isIpad()) {
+    NavigationContainerRef.reset({
+      index: 0,
+      routes: [{ name: "home" }],
+    });
+    getDefaultStore().set(RoomAtoms.selectedRoomId, roomId);
+    return;
+  }
+
+  NavigationContainerRef.reset({
+    index: 1,
+    routes: [{ name: "home" }, { name: "room", params: { roomId } }],
+  });
+};

+import { animateParallel, Block } from "@prototyp/skeletor";
+import { useAtomValue } from "jotai";
+import { MatrixAtoms } from "modules/matrix-client/jotai";
+import { NavigationContainerRef } from "modules/navigation/const";
+import { NavigationAtoms } from "modules/navigation/jotai";
+import { type PropsWithChildren, useEffect, useRef, useState } from "react";
+import { Easing } from "react-native";
+import { isIpad } from "shared/utils";
+import { IpadSidebar } from "../ipad-sidebar";
+
+const SIDEBAR_WIDTH = 440;
+
+const FULLSCREEN_ROUTES: (keyof Routes)[] = [
+  "splash",
+  "welcome",
+  "login",
+  "onboarding",
+  "notificationsIntro",
+  "serverSelect",
+  "confirmIdentity",
+  "cantConfirm",
+  "verifyUser",
+  "verifyWithDevice",
+  "enterRecoveryKey",
+  "oidcResetWebView",
+];
+
+export const IpadLayoutWrapper: React.FC<PropsWithChildren> = ({
+  children,
+}) => {
+  const ipad = isIpad();
+  const splashFinished = useAtomValue(NavigationAtoms.splashFinished);
+  const matrixClient = useAtomValue(MatrixAtoms.matrixClient);
+  const [currentRoute, setCurrentRoute] = useState<string | undefined>(
+    undefined,
+  );
+  const currentRouteRef = useRef<string | undefined>(undefined);
+
+  const showSidebar =
+    ipad &&
+    splashFinished &&
+    matrixClient !== null &&
+    !FULLSCREEN_ROUTES.includes(currentRoute as keyof Routes);
+
+  const transition = useRef(
+    animateParallel(
+      { marginLeft: [0, SIDEBAR_WIDTH] },
+      { duration: 350, easing: Easing.out(Easing.cubic), native: false },
+    ),
+  ).current;
+
+  const routeTimeoutRef = useRef<ReturnType<typeof setTimeout> | null>(null);
+
+  const updateRoute = (newRoute: string | undefined) => {
+    currentRouteRef.current = newRoute;
+    setCurrentRoute(newRoute);
+  };
+
+  useEffect(() => {
+    if (!NavigationContainerRef?.isReady()) return;
+    NavigationContainerRef.addListener("state", () => {
+      const newRoute = NavigationContainerRef?.getCurrentRoute()?.name;
+      const prevRoute = currentRouteRef.current;
+
+      if (routeTimeoutRef.current) clearTimeout(routeTimeoutRef.current);
+
+      const isLeavingFullscreen =
+        FULLSCREEN_ROUTES.includes(prevRoute as keyof Routes) &&
+        !FULLSCREEN_ROUTES.includes(newRoute as keyof Routes);
+
+      if (isLeavingFullscreen) {
+        // Delay until the stack pop animation completes (~350ms) so the
+        // marginLeft shift doesn't conflict with the screen transition.
+        routeTimeoutRef.current = setTimeout(() => updateRoute(newRoute), 380);
+      } else {
+        updateRoute(newRoute);
+      }
+    });
+  }, [NavigationContainerRef?.isReady()]);
+
+  useEffect(() => {
+    if (showSidebar) {
+      setTimeout(() => transition.start(), 200);
+    } else transition.reverse();
+  }, [showSidebar]);
+
+  return (
+    <Block flex={1}>
+      <Block flex={1} animations={transition.animations}>
+        {children}
+      </Block>
+      {showSidebar && <IpadSidebar />}
+    </Block>
+  );
+};
